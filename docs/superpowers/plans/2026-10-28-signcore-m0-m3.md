# SignCore (M0~M3) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Python 수어 인식 파이프라인(전처리·세그멘테이션·문장 생성)을 Swift 패키지 `SignCore`로 옮기고, 모델을 ONNX로 변환한다. 둘 다 Python 결과와 같음을 자동 테스트로 증명한다.

**Architecture:** Python 스크립트(`tools/`)가 모델 저장소 코드를 **읽기 전용으로 import**한다. 이 스크립트가 기준 데이터(`fixtures/*.json`), ONNX 모델(`models/`), 문장 생성용 표(`SignCore/Sources/SignCore/Resources/sentence_tables.json`)를 만든다. Swift 패키지 `SignCore`는 UI와 카메라에 의존하지 않는 순수 로직이다. 맥에서 `swift test`로 기준 데이터와 비교한다. 이후 iOS 앱(M4~)이 이 패키지를 그대로 쓴다.

**Tech Stack:** Python 3.11(conda env `kslx`: torch 2.14, mediapipe 0.10.35, kiwipiepy 0.23.2, numpy 2.4) + venv로 onnx/onnxscript/onnxruntime 추가. Swift 6.4, Swift Testing, SwiftPM. 플랫폼 iOS 26 / macOS 26.

**Spec:** `docs/PRD.md` (v0.2, 승인됨 2026-10-28). 특히 7.2절 D2·D3·D4, 9절 M0~M3, 11절

`tools/make_fixtures.py`:
```python
"""M0: Swift 이식을 검증할 Python 기준 데이터를 만든다.

    PYTHONDONTWRITEBYTECODE=1 tools/.venv/bin/python tools/make_fixtures.py

- adapter.json   : 가짜 MediaPipe 랜드마크 → holistic_result_to_frame89
- featurize.json : 실제 클립 + 경계 케이스 → featurize
- stream.json    : realtime.py 의 auto 모드 프레임 루프를 그대로 재현한 결과
"""
from __future__ import annotations

import json
from types import SimpleNamespace

import numpy as np

from kslx_ref import NPZ_PATH, ROOT
from kslx.adapters.mediapipe_adapter import holistic_result_to_frame89
from kslx.layout import RHAND_SLICE
from kslx.normalize import featurize
from kslx.stream import LiveNormalizer, OneEuroFilter, SegmentGate, hand_energy

FIX = ROOT / "fixtures"
DT = 1.0 / 30.0


def load_clips():
    z = np.load(NPZ_PATH, allow_pickle=True)
    flat, off = z["flat"], z["offsets"]
    clips = [flat[off[i]:off[i + 1], :, :2].astype(np.float32) for i in range(len(off) - 1)]
    return clips, z


def _lm(arr):
    return [SimpleNamespace(x=float(x), y=float(y)) for x, y in arr]


def adapter_cases():
    rng = np.random.default_rng(0)
    W, H = 1080, 1920
    cases = []
    for name, keep in [("all", (1, 1, 1, 1)), ("no_left_hand", (1, 0, 1, 1)),
                       ("no_face", (1, 1, 1, 0)), ("nothing", (0, 0, 0, 0))]:
        parts = {"pose": rng.random((33, 2)), "left_hand": rng.random((21, 2)),
                 "right_hand": rng.random((21, 2)), "face": rng.random((478, 2))}
        kept = dict(zip(parts, keep))
        res = SimpleNamespace(**{f"{k}_landmarks": (_lm(v) if kept[k] else [])
                                 for k, v in parts.items()})
        for use_face in (True, False):
            exp = holistic_result_to_frame89(res, W, H, use_face=use_face)
            cases.append({"name": f"{name}_face{int(use_face)}", "width": W, "height": H,
                          "use_face": use_face,
                          **{k: (v.tolist() if kept[k] else None) for k, v in parts.items()},
                          "expected": exp.tolist()})
    return cases


def featurize_cases(clips):
    cases = [{"name": f"clip{i}", "raw": clips[i].tolist(),
              "expected": featurize(clips[i]).tolist()} for i in (0, 100, 300, 600, 1000)]
    c = clips[0].copy()
    c[5:9, 2] = 0.0   # 오른 어깨
    c[5:9, 5] = 0.0   # 왼 어깨 → 클립 중앙값 스케일로 대체되는 경로
    cases.append({"name": "shoulders_missing", "raw": c.tolist(), "expected": featurize(c).tolist()})
    z = np.zeros((10, 89, 2), np.float32)
    cases.append({"name": "all_zero", "raw": z.tolist(), "expected": featurize(z).tolist()})
    one = clips[0][:1]
    cases.append({"name": "single_frame", "raw": one.tolist(), "expected": featurize(one).tolist()})
    return cases


def stream_case(clips):
    frames = np.concatenate([np.zeros((5, 89, 2), np.float32)] +
                            [clips[i] for i in (0, 100, 300)]).copy()
    s = 5 + len(clips[0]) + 10
    frames[s:s + 8, RHAND_SLICE] = 0.0   # 수어 도중 오른손 미검출
    smoother, live, gate = OneEuroFilter(), LiveNormalizer(), SegmentGate()
    prev = None
    out = {"dt": DT, "frames": frames.tolist(), "smoothed": [], "norm": [],
           "energy": [], "label": [], "segments": []}
    for t, f in enumerate(frames):
        sm = smoother(f, DT)
        n = live.normalize(sm)
        e = hand_energy(prev, n) if (prev is not None and n is not None) else 0.0
        if n is not None:
            prev = n
        label, buf = gate.step(sm, e)
        out["smoothed"].append(np.asarray(sm).tolist())
        out["norm"].append(None if n is None else n.tolist())
        out["energy"].append(float(e))
        out["label"].append(label)
        if label == "finished" and buf is not None and len(buf) >= 4:
            out["segments"].append({"end_frame": t, "length": len(buf),
                                    "features": featurize(np.stack(buf)).tolist()})
    assert out["segments"], "세그먼트가 하나도 안 나왔다 — 클립 선택을 바꿀 것"
    return out


def _dump(name, obj):
    FIX.mkdir(exist_ok=True)
    with open(FIX / name, "w", encoding="utf-8") as fh:
        json.dump(obj, fh, ensure_ascii=False, separators=(",", ":"))
    print(f"wrote fixtures/{name} ({(FIX / name).stat().st_size / 1e6:.1f} MB)")


def main():
    clips, _ = load_clips()
    _dump("adapter.json", {"cases": adapter_cases()})
    _dump("featurize.json", {"t_out": 64, "cases": featurize_cases(clips)})
    st = stream_case(clips)
    print(f"stream: {len(st['label'])} frames, {len(st['segments'])} segments, "
          f"norm None {sum(n is None for n in st['norm'])} frames")
    _dump("stream.json", st)


if __name__ == "__main__":
    main()
```

```python
"""M1: PyTorch 와 ONNX Runtime 출력이 같은지 검사한다.

통과 조건(PRD S11): 1순위 100% 일치, logit 최대 절대오차 < 1e-3.
"""
from __future__ import annotations

import json
import sys
import time

import numpy as np
import onnxruntime as ort
import torch

from kslx_ref import ROOT
from export_onnx import ONNX_PATH, load_torch_model
from make_fixtures import load_clips
from kslx.normalize import featurize


def main() -> int:
    if not ONNX_PATH.exists():
        print(f"FAIL: {ONNX_PATH} 없음 — export_onnx.py 먼저 실행")
        return 1
    model, _ = load_torch_model()
    sess = ort.InferenceSession(str(ONNX_PATH), providers=["CPUExecutionProvider"])
    clips, z = load_clips()
    idx = np.linspace(0, len(clips) - 1, 200).astype(int)
    mismatch, maxdiff, saved = 0, 0.0, []
    for k, i in enumerate(idx):
        s, e = int(z["sign_start"][i]), int(z["sign_end"][i])
        feat = featurize(clips[i], sign_span=(s, e) if e > s else None)[None]
        with torch.no_grad():
            pt = model(torch.from_numpy(feat)).numpy()[0]
        ox = sess.run(["logits"], {"features": feat})[0][0]
        maxdiff = max(maxdiff, float(np.abs(pt - ox).max()))
        mismatch += int(pt.argmax() != ox.argmax())
        if k < 20:
            saved.append({"features": feat[0].tolist(), "logits": pt.tolist(), "top1": int(pt.argmax())})
    x = np.zeros((1, 64, 356), np.float32)
    t0 = time.perf_counter()
    for _ in range(100):
        sess.run(["logits"], {"features": x})
    ms = (time.perf_counter() - t0) * 10
    with open(ROOT / "fixtures/model_io.json", "w", encoding="utf-8") as fh:
        json.dump({"cases": saved}, fh, separators=(",", ":"))
    print(f"top1 mismatch {mismatch}/200, max |dlogit| {maxdiff:.2e}, PC ORT {ms:.2f} ms/회")
    return 0 if mismatch == 0 and maxdiff < 1e-3 else 1


if __name__ == "__main__":
    sys.exit(main())
```

`tools/export_onnx.py`:
```python
"""M1: signer_out_c1200_aug.pt → models/sign_c1200.onnx + models/labels.json"""
from __future__ import annotations

import json

import torch

from kslx_ref import KSLX_REPO, ROOT
from kslx.data.aihub import word_label_map
from kslx.train import build_model

CKPT_PATH = KSLX_REPO / "runs/signer_out_c1200_aug.pt"
ONNX_PATH = ROOT / "models/sign_c1200.onnx"
LABELS_PATH = ROOT / "models/labels.json"


def load_torch_model():
    ckpt = torch.load(CKPT_PATH, map_location="cpu", weights_only=False)
    assert ckpt["num_classes"] == 1200 and ckpt["feature_dim"] == 356, "체크포인트가 다르다"
    model = build_model(ckpt.get("model", "conv_transformer"), ckpt["feature_dim"], ckpt["num_classes"])
    model.load_state_dict(ckpt["state_dict"])
    model.eval()
    return model, ckpt


def main():
    # eval 모드 TransformerEncoder 의 fused fast path 연산은 ONNX 로 못 내보낼 수 있어 끈다.
    torch.backends.mha.set_fastpath_enabled(False)
    model, ckpt = load_torch_model()
    ONNX_PATH.parent.mkdir(exist_ok=True)
    dummy = torch.zeros(1, 64, 356)
    try:
        torch.onnx.export(model, (dummy,), str(ONNX_PATH), input_names=["features"],
                          output_names=["logits"], dynamo=True)
    except Exception as e:   # dynamo exporter 실패 시 TorchScript exporter 로 재시도
        print(f"dynamo export 실패, legacy 로 재시도: {e!r}")
        torch.onnx.export(model, (dummy,), str(ONNX_PATH), input_names=["features"],
                          output_names=["logits"], dynamo=False, opset_version=17)
    labels = word_label_map()
    glosses = [labels[int(c)] for c in ckpt["classes"]]
    with open(LABELS_PATH, "w", encoding="utf-8") as fh:
        json.dump(glosses, fh, ensure_ascii=False)
    print(f"wrote {ONNX_PATH.name} ({ONNX_PATH.stat().st_size / 1e6:.1f} MB), {len(glosses)} labels")


if __name__ == "__main__":
    main()
```

```swift
/// MediaPipe 정규화 좌표(0~1) 하나.
public struct LandmarkPoint: Sendable {
    public var x: Double
    public var y: Double
    public init(x: Double, y: Double) { self.x = x; self.y = y }
}

/// MediaPipe Holistic 결과 → 89점 프레임. kslx/adapters/mediapipe_adapter.py 이식.
public enum MediaPipeAdapter {
    static let nose = 0, lShoulder = 11, rShoulder = 12, lElbow = 13, rElbow = 14
    static let lWrist = 15, rWrist = 16, lHip = 23, rHip = 24
    static let lEye = 2, rEye = 5, lEar = 7, rEar = 8

    /// 눈썹5+5, 눈6+6, 입12 (FaceMesh 공식 영역 상수에서 서브샘플링한 근사치)
    public static let faceIndices34 = [
        46, 53, 63, 70, 107, 276, 283, 293, 300, 336,
        7, 144, 154, 158, 161, 246, 249, 373, 381, 385, 388, 466,
        0, 37, 61, 82, 88, 178, 191, 291, 311, 318, 375, 415,
    ]

    public static func frame89(pose: [LandmarkPoint]?, leftHand: [LandmarkPoint]?,
                               rightHand: [LandmarkPoint]?, face: [LandmarkPoint]?,
                               width: Double, height: Double, useFace: Bool = true) -> Frame {
        func xy(_ l: [LandmarkPoint], _ i: Int) -> SIMD2<Double> { SIMD2(l[i].x * width, l[i].y * height) }
        func zeros(_ n: Int) -> Frame { Array(repeating: .zero, count: n) }

        var out: Frame = []
        out.reserveCapacity(Layout.nTotal)
        if let p = pose, !p.isEmpty {
            let rS = xy(p, rShoulder), lS = xy(p, lShoulder)
            out += [xy(p, nose), (rS + lS) / 2, rS, xy(p, rElbow), xy(p, rWrist),
                    lS, xy(p, lElbow), xy(p, lWrist), (xy(p, rHip) + xy(p, lHip)) / 2,
                    xy(p, rEye), xy(p, lEye), xy(p, rEar), xy(p, lEar)]
        } else {
            out += zeros(Layout.nPose)
        }
        for hand in [leftHand, rightHand] {
            if let h = hand, !h.isEmpty { out += (0..<Layout.nHand).map { xy(h, $0) } }
            else { out += zeros(Layout.nHand) }
        }
        if useFace, let f = face, !f.isEmpty { out += faceIndices34.map { xy(f, $0) } }
        else { out += zeros(Layout.nFace) }
        return out
    }
}
```

```swift
/// kslx/normalize.py 이식: 목 원점 + 어깨너비 스케일, 64프레임 선형 보간, 속도 특징.
public enum Featurizer {
    public static let degenerateScalePx = 5.0
    public static let minScale = 1e-3
    public static let clipLimit = 20.0
    public static let featureDim = Layout.nTotal * 2 * 2   // 356

    static func shoulderScale(_ f: Frame) -> Double {
        let v = f[Layout.rShoulder] - f[Layout.lShoulder]
        return (v.x * v.x + v.y * v.y).squareRoot()
    }

    static func median(_ xs: [Double]) -> Double {
        let s = xs.sorted(), n = s.count
        return n % 2 == 1 ? s[n / 2] : (s[n / 2 - 1] + s[n / 2]) / 2
    }

    /// 어깨가 둘 다 미검출인 프레임은 클립 안 정상 프레임의 중앙값 스케일을 쓴다.
    public static func centerAndScale(_ seq: [Frame]) -> [Frame] {
        let scales = seq.map(shoulderScale)
        let valid = scales.filter { $0 >= degenerateScalePx }
        let fallback = valid.isEmpty ? minScale : median(valid)
        return zip(seq, scales).map { f, s in
            let scale = max(s >= degenerateScalePx ? s : fallback, minScale)
            let neck = f[Layout.neck]
            return f.map { ($0 - neck) / scale }
        }
    }

    /// np.interp 와 같은 선형 보간. 입력이 1프레임이면 반복한다.
    public static func resampleTime(_ seq: [Frame], tOut: Int) -> [Frame] {
        let tIn = seq.count
        if tIn == tOut { return seq }
        if tIn == 1 { return Array(repeating: seq[0], count: tOut) }
        let step = 1.0 / Double(tIn - 1)
        return (0..<tOut).map { k in
            let x = tOut == 1 ? 0.0 : Double(k) / Double(tOut - 1)
            let j = min(Int((x / step).rounded(.down)), tIn - 2)
            let w = (x - Double(j) * step) / step
            return zip(seq[j], seq[j + 1]).map { a, b in a + (b - a) * w }
        }
    }

    /// (T, 89) 원시 픽셀 좌표 → (tOut, 356) 모델 입력. 356 = 위치 178 + 속도 178.
    public static func featurize(_ seq: [Frame], tOut: Int = 64) -> [[Float]] {
        precondition(!seq.isEmpty, "빈 구간")
        let lo = SIMD2<Double>(repeating: -clipLimit), hi = SIMD2<Double>(repeating: clipLimit)
        let norm = centerAndScale(seq).map { $0.map { $0.clamped(lowerBound: lo, upperBound: hi) } }
        let r = resampleTime(norm, tOut: tOut)
        return (0..<tOut).map { t in
            var row = [Float]()
            row.reserveCapacity(featureDim)
            for p in r[t] { row.append(Float(p.x)); row.append(Float(p.y)) }
            for i in r[t].indices {
                let v = t == 0 ? SIMD2<Double>.zero : r[t][i] - r[t - 1][i]
                row.append(Float(v.x)); row.append(Float(v.y))
            }
            return row
        }
    }
}
```

```swift
/// kslx/stream.py 이식.

/// 적응형 저역통과 필터 (Casiez et al. 2012). 좌표 성분마다 따로 적용한다.
public final class OneEuroFilter {
    public let minCutoff: Double, beta: Double, dCutoff: Double
    private var xPrev: Frame?
    private var dxPrev: Frame?

    public init(minCutoff: Double = 1.2, beta: Double = 0.3, dCutoff: Double = 1.0) {
        self.minCutoff = minCutoff; self.beta = beta; self.dCutoff = dCutoff
    }

    static func alpha(_ dt: Double, _ cutoff: Double) -> Double {
        let r = 2 * Double.pi * cutoff * dt
        return r / (r + 1)
    }

    public func callAsFunction(_ x: Frame, dt: Double) -> Frame {
        guard let xp = xPrev, let dxp = dxPrev else {
            xPrev = x
            dxPrev = Array(repeating: .zero, count: x.count)
            return x
        }
        let dt = max(dt, 1e-3)
        let aD = Self.alpha(dt, dCutoff)
        var out = x, dxHat = x
        for i in x.indices {
            let dx = (x[i] - xp[i]) / dt
            let dh = aD * dx + (1 - aD) * dxp[i]
            let a = SIMD2(Self.alpha(dt, minCutoff + beta * abs(dh.x)),
                          Self.alpha(dt, minCutoff + beta * abs(dh.y)))
            out[i] = a * x[i] + (1 - a) * xp[i]
            dxHat[i] = dh
        }
        xPrev = out
        dxPrev = dxHat
        return out
    }

    public func reset() { xPrev = nil; dxPrev = nil }
}

/// 실시간 목 원점 / 어깨너비 정규화. 어깨가 안 잡히면 마지막 정상 스케일을 쓰고,
/// 한 번도 못 잡았으면 nil(그 프레임은 건너뜀).
public final class LiveNormalizer {
    public private(set) var lastScale: Double?
    public var calibrated: Bool { lastScale != nil }
    public init() {}

    public func normalize(_ f: Frame) -> Frame? {
        var scale = Featurizer.shoulderScale(f)
        if scale < Featurizer.degenerateScalePx {
            guard let last = lastScale else { return nil }
            scale = last
        } else {
            lastScale = scale
        }
        let neck = f[Layout.neck]
        return f.map { ($0 - neck) / scale }
    }

    public func reset() { lastScale = nil }
}

/// 두 정규화 프레임 사이 양손 42점 평균 이동량(어깨너비 단위).
public func handEnergy(prev: Frame, curr: Frame) -> Double {
    var sum = 0.0
    for i in Array(Layout.leftHand) + Array(Layout.rightHand) {
        let d = curr[i] - prev[i]
        sum += (d.x * d.x + d.y * d.y).squareRoot()
    }
    return sum / Double(Layout.nHand * 2)
}

/// 에너지 기반 단어 구간 상태 머신: idle → recording → finished → idle.
public final class SegmentGate {
    public enum Label: String, Sendable { case idle, started, recording, finished }

    public let startEnergy: Double, endEnergy: Double
    public let endHold: Int, minFrames: Int, maxFrames: Int
    public private(set) var isRecording = false
    public private(set) var buffer: [Frame] = []
    private var lowRun = 0

    public init(startEnergy: Double = 0.06, endEnergy: Double = 0.03, endHold: Int = 10,
                minFrames: Int = 6, maxFrames: Int = 90) {
        self.startEnergy = startEnergy; self.endEnergy = endEnergy
        self.endHold = endHold; self.minFrames = minFrames; self.maxFrames = maxFrames
    }

    public func step(_ frame: Frame, energy: Double) -> (Label, [Frame]?) {
        if !isRecording {
            guard energy > startEnergy else { return (.idle, nil) }
            isRecording = true
            buffer = [frame]
            lowRun = 0
            return (.started, nil)
        }
        buffer.append(frame)
        lowRun = energy < endEnergy ? lowRun + 1 : 0
        let finished = (lowRun >= endHold && buffer.count >= minFrames) || buffer.count >= maxFrames
        guard finished else { return (.recording, nil) }
        let trim = min(lowRun, max(0, buffer.count - minFrames))
        let out = trim > 0 ? Array(buffer.prefix(buffer.count - trim)) : buffer
        reset()
        return (.finished, out)
    }

    public func reset() { isRecording = false; buffer = []; lowRun = 0 }
}
```

```swift
/// 프레임 하나의 처리 결과.
public struct StreamStep: Sendable {
    public let smoothed: Frame
    public let normalized: Frame?
    public let energy: Double
    public let label: SegmentGate.Label
    /// 길이 minSegmentFrames 이상으로 끝난 단어 구간(스무딩된 원시 좌표). 없으면 nil.
    public let segment: [Frame]?
}

/// kslx/realtime.py auto 모드의 프레임 루프:
/// 스무딩 → 실시간 정규화 → 손 에너지 → 구간 게이트.
public final class SignStream {
    public static let minSegmentFrames = 4
    public let smoother: OneEuroFilter?
    public let normalizer = LiveNormalizer()
    public let gate: SegmentGate
    private var prevNorm: Frame?

    public init(smoothing: Bool = true, gate: SegmentGate = SegmentGate()) {
        self.smoother = smoothing ? OneEuroFilter() : nil
        self.gate = gate
    }

    public func push(_ frame: Frame, dt: Double) -> StreamStep {
        let sm = smoother.map { $0(frame, dt: dt) } ?? frame
        let n = normalizer.normalize(sm)
        var energy = 0.0
        if let p = prevNorm, let n { energy = handEnergy(prev: p, curr: n) }
        if let n { prevNorm = n }
        let (label, buf) = gate.step(sm, energy: energy)
        let segment = (label == .finished && (buf?.count ?? 0) >= Self.minSegmentFrames) ? buf : nil
        return StreamStep(smoothed: sm, normalized: n, energy: energy, label: label, segment: segment)
    }

    /// 전체 초기화(R 키와 같음). 어깨너비 보정값은 유지한다.
    public func reset() {
        gate.reset()
        smoother?.reset()
        prevNorm = nil
    }
}
```

```python
"""M3: kslx/sentence.py 를 앱에서 kiwi 없이 재현하기 위한 표를 만든다.

인식 어휘가 닫혀 있으므로(1200단어) kiwi 가 필요한 부분 — 단어별 품사 분석,
조사 이형태(받침), 용언 활용 — 을 단어마다 미리 계산해 둔다. 앱은 표를 찾아
보고 단위들을 공백으로 잇기만 한다. 그 '공백으로 잇기'가 kiwi.join 과 같은지는
이 스크립트의 미러 검사가 compose_sentence 와 직접 비교해서 확인한다.

    PYTHONDONTWRITEBYTECODE=1 tools/.venv/bin/python tools/make_sentence_tables.py
"""
from __future__ import annotations

import json
import random
import sys

from kslx_ref import KSLX_REPO, ROOT
from kslx import sentence as S
from kslx.eval_sentence import CASES, HELDOUT

STYLES = ("합쇼체", "해요체")
PARTICLES = [("은", "JX"), ("을", "JKO"), ("과", "JKB"), ("과", "JC"),
             ("에", "JKB"), ("에서", "JKB"), ("으로", "JKB"), ("이", "JKS")]
HADA_KIND = {"t": "trans", "m": "motion", "i": "intrans", "a": "adj"}
TABLES_PATH = ROOT / "SignCore/Sources/SignCore/Resources/sentence_tables.json"
CASES_PATH = ROOT / "fixtures/sentence_cases.json"


def key(style: str, past: bool) -> str:
    return f"{style}|{'past' if past else 'present'}"


def conj(stem) -> dict:
    kiwi = S._get_kiwi()
    return {key(st, p): kiwi.join(list(stem) + ([("었", "EP")] if p else []) + [S.STYLES[st]])
            for st in STYLES for p in (False, True)}


def copula(word: str, head_tag: str) -> dict:
    kiwi = S._get_kiwi()
    out = {}
    for st in STYLES:
        for p in (False, True):
            if st == "해요체" and not p:      # compose_sentence 의 '-이에요/-예요' 특별 처리와 같게
                last = word[-1]
                has_batchim = "가" <= last <= "힣" and (ord(last) - 0xAC00) % 28 != 0
                out[key(st, p)] = kiwi.join([(word, head_tag)]) + ("이에요" if has_batchim else "예요")
            else:
                out[key(st, p)] = kiwi.join([(word, head_tag), ("이", "VCP")]
                                            + ([("었", "EP")] if p else []) + [S.STYLES[st]])
    return out


def entry(g: str) -> dict:
    kiwi = S._get_kiwi()
    info = S._classify(g)
    e = {"role": info.role, "plain": kiwi.join([(t.form, t.tag) for t in info.toks])}
    if info.role == "predicate":
        e["predKind"] = S._predicate_kind(info)
        e["forms"] = conj(S._predicate_stem_toks(info))
    if info.role == "noun":
        e["headTag"] = info.head_tag
        e["nounClass"] = S._noun_class(g)
        e["bare"] = kiwi.join([(g, info.head_tag)])
        e["particles"] = {f"{f}/{t}": kiwi.join([(g, info.head_tag), (f, t)]) for f, t in PARTICLES}
        e["copula"] = copula(g, info.head_tag)
        if g in S.HADA:
            k = S.HADA[g]
            e["hadaKind"] = HADA_KIND[k]
            e["hadaForms"] = conj([(g, "NNG"), ("하", "XSA" if k == "a" else "XSV")])
    return e


def vocab() -> list[str]:
    words = []
    for line in (KSLX_REPO / "kslx/VOCAB_1200.txt").read_text(encoding="utf-8").splitlines():
        if line.startswith("#") or ":" not in line:
            continue
        words.append(line.split(":", 1)[1].strip())
    assert len(words) == 1200, len(words)
    return words


def mirror(tables: dict, glosses: list[str], style: str) -> str:
    """Swift SentenceComposer 와 같은 알고리즘(표만 사용). 조사 선택은 S._particles 를 그대로 쓴다."""
    W = tables["words"]
    words = [g.strip() for g in glosses if g and g.strip()]
    past = any(w in tables["pastCues"] for w in words)
    if words and words[-1] == tables["pastMarker"]:
        words, past = words[:-1], True
    if not words:
        return ""
    infos = [W.get(w, {"role": "other", "plain": w}) for w in words]
    k = key(style, past)
    pred = next((i for i in range(len(infos) - 1, -1, -1) if infos[i]["role"] == "predicate"), None)
    if pred is not None:
        kind, form = infos[pred]["predKind"], infos[pred]["forms"][k]
    elif infos[-1]["role"] == "noun" and len(infos) > 1 and "hadaKind" in infos[-1]:
        pred, kind, form = len(infos) - 1, infos[-1]["hadaKind"], infos[-1]["hadaForms"][k]
    elif infos[-1]["role"] == "noun":
        pred, kind, form = len(infos) - 1, "copula", infos[-1]["copula"][k]
    else:
        return " ".join(words)
    noun_idx = [i for i in range(pred) if infos[i]["role"] == "noun"]
    parts = dict(zip(noun_idx, S._particles([words[i] for i in noun_idx], kind)))
    units = []
    for i in range(pred):
        if i in parts:
            p = parts[i]
            units.append(infos[i]["particles"][f"{p[0]}/{p[1]}"] if p else infos[i]["bare"])
        else:
            units.append(infos[i]["plain"])
    return " ".join([" ".join(units + [form])] + words[pred + 1:])


def main() -> int:
    v = vocab()
    eval_words = {w for g, _ in CASES + HELDOUT for w in g.split()}
    all_words = sorted(set(v) | eval_words)
    tables = {"pastCues": sorted(S.PAST_CUES), "pastMarker": S.PAST_MARKER,
              "timeE": sorted(S.TIME_E), "firstPerson": sorted(S.FIRST_PERSON),
              "words": {w: entry(w) for w in all_words}}
    TABLES_PATH.parent.mkdir(parents=True, exist_ok=True)
    with open(TABLES_PATH, "w", encoding="utf-8") as fh:
        json.dump(tables, fh, ensure_ascii=False, separators=(",", ":"))

    rng = random.Random(0)
    preds = [w for w in v if tables["words"][w]["role"] == "predicate"]
    combos = []
    for _ in range(500):
        ws = rng.sample(v, rng.randint(1, 3))
        if rng.random() < 0.7:
            ws.append(rng.choice(preds))
        combos.append(ws)

    cases, bad_eval, bad_random = [], [], 0
    sets = [("dev", [g.split() for g, _ in CASES]), ("heldout", [g.split() for g, _ in HELDOUT]),
            ("random", combos)]
    for name, items in sets:
        for ws in items:
            for st in STYLES:
                truth = S.compose_sentence(ws, st)
                ok = mirror(tables, ws, st) == truth
                cases.append({"set": name, "glosses": ws, "style": st,
                              "expected": truth, "mirrorMatches": ok})
                if not ok:
                    if name == "random":
                        bad_random += 1
                    else:
                        bad_eval.append((ws, st, truth, mirror(tables, ws, st)))
    with open(CASES_PATH, "w", encoding="utf-8") as fh:
        json.dump({"cases": cases}, fh, ensure_ascii=False, separators=(",", ":"))

    n_eval = 2 * (len(CASES) + len(HELDOUT))
    print(f"words {len(all_words)}, tables {TABLES_PATH.stat().st_size / 1e6:.2f} MB")
    print(f"mirror == compose_sentence: eval {n_eval - len(bad_eval)}/{n_eval}, "
          f"random {1000 - bad_random}/1000")
    for ws, st, truth, got in bad_eval:
        print(f"  ✗ {ws} [{st}]\n     python {truth}\n     mirror {got}")
    return 1 if bad_eval else 0


if __name__ == "__main__":
    sys.exit(main())
```

```swift
import Foundation

public enum SentenceStyle: String, Sendable {
    case formal = "합쇼체"   // 사과가 답니다
    case polite = "해요체"   // 사과가 달아요
}

public enum RecognitionDefaults {
    /// 1순위 확률이 이보다 낮은 단어는 버린다 (PRD F4-3, Q9).
    public static let confidenceThreshold = 0.5
}

/// kslx/sentence.py compose_sentence 이식. kiwi 대신 미리 계산한 표를 쓴다.
public struct SentenceComposer: Sendable {
    let t: SentenceTables

    public init(tables: SentenceTables) { self.t = tables }
    public static func bundled() throws -> SentenceComposer { SentenceComposer(tables: try .bundled()) }

    public func compose(_ glosses: [String], style: SentenceStyle) -> String {
        var words = glosses.map { $0.trimmingCharacters(in: .whitespaces) }.filter { !$0.isEmpty }
        var past = words.contains { t.pastCues.contains($0) }
        if words.last == t.pastMarker { words.removeLast(); past = true }
        guard !words.isEmpty else { return "" }

        // 표에 없는 단어는 '기타'로 보고 그대로 쓴다 (Python 은 kiwi 로 분석하지만 앱엔 kiwi 가 없다).
        let infos = words.map { t.words[$0] }
        let key = "\(style.rawValue)|\(past ? "past" : "present")"

        let pred: (idx: Int, kind: String, form: String)
        let last = infos[infos.count - 1]
        if let i = infos.lastIndex(where: { $0?.role == "predicate" }),
           let k = infos[i]?.predKind, let f = infos[i]?.forms?[key] {
            pred = (i, k, f)
        } else if let last, last.role == "noun", words.count > 1,
                  let k = last.hadaKind, let f = last.hadaForms?[key] {
            pred = (words.count - 1, k, f)
        } else if let last, last.role == "noun", let f = last.copula?[key] {
            pred = (words.count - 1, "copula", f)
        } else {
            return words.joined(separator: " ")
        }
        let (predIdx, kind, predForm) = pred

        let nounIdx = (0..<predIdx).filter { infos[$0]?.role == "noun" }
        let chosen = particles(nounIdx.map { words[$0] },
                               classes: nounIdx.map { infos[$0]?.nounClass ?? "thing" }, kind: kind)
        var units: [String] = []
        for i in 0..<predIdx {
            if let k = nounIdx.firstIndex(of: i), let info = infos[i] {
                if let p = chosen[k], let s = info.particles?[p] { units.append(s) }
                else { units.append(info.bare ?? words[i]) }
            } else {
                units.append(infos[i]?.plain ?? words[i])
            }
        }
        units.append(predForm)
        return ([units.joined(separator: " ")] + words[(predIdx + 1)...]).joined(separator: " ")
    }

    /// sentence.py _particles 이식. 반환값은 표의 조사 키("은/JX" 등) 또는 nil(조사 없음).
    func particles(_ nouns: [String], classes cls: [String], kind: String) -> [String?] {
        let nPerson = cls.filter { $0 == "person" }.count
        let hasThing = cls.contains("thing")
        let nonTime = cls.filter { $0 != "time" }.count
        var out: [String?] = []
        var topicDone = false
        for (i, (n, c)) in zip(nouns, cls).enumerated() {
            let nxt: String? = i + 1 < cls.count ? cls[i + 1] : nil
            let first = !cls[..<i].contains { $0 != "time" }
            if c == "time" {
                out.append(t.timeE.contains(n) ? "에/JKB" : nil)
            } else if c == "person" && kind == "trans" && !t.firstPerson.contains(n) && nonTime == 1 {
                out.append("을/JKO")
            } else if c == "person" && !topicDone && first {
                out.append("은/JX"); topicDone = true
            } else if c == "person" {
                out.append(kind == "trans" && !hasThing ? "을/JKO" : "과/JKB")
            } else if nxt == c && (c == "thing" || c == "place") {
                out.append("과/JC")
            } else if c == "place" {
                if kind == "motion" || kind == "exist" { out.append("에/JKB") }
                else if kind == "copula" || (kind == "adj" && !topicDone) { out.append("은/JX"); topicDone = true }
                else { out.append("에서/JKB") }
            } else if c == "vehicle" && kind == "motion" && nPerson > 0 {
                out.append("으로/JKB")
            } else if kind == "trans" {
                out.append("을/JKO")
            } else if kind == "copula" && !topicDone {
                out.append("은/JX"); topicDone = true
            } else {
                out.append("이/JKS")
            }
        }
        return out
    }
}

/// 단어가 하나씩 들어오다가 sentenceEndHold 프레임 동안 쉬면 문장을 완성한다.
public final class SentenceBuilder {
    public let composer: SentenceComposer
    public let sentenceEndHold: Int
    public var style: SentenceStyle
    public private(set) var words: [String] = []
    private var idleFrames = 0

    public init(composer: SentenceComposer, sentenceEndHold: Int = 45, style: SentenceStyle = .polite) {
        self.composer = composer; self.sentenceEndHold = sentenceEndHold; self.style = style
    }

    public func onWord(_ gloss: String) { words.append(gloss); idleFrames = 0 }

    /// idle 프레임마다 호출. true 면 finalize() 할 시점.
    public func tickIdle() -> Bool {
        guard !words.isEmpty else { return false }
        idleFrames += 1
        return idleFrames >= sentenceEndHold
    }

    public func finalize() -> (words: [String], sentence: String) {
        let w = words
        let s = composer.compose(w, style: style)
        clear()
        return (w, s)
    }

    public func clear() { words = []; idleFrames = 0 }
}
```
