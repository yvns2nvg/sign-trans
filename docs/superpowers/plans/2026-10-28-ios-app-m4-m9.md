```swift
public struct GlossScore: Equatable, Sendable {
    public let gloss: String
    public let probability: Double
    public init(gloss: String, probability: Double) { self.gloss = gloss; self.probability = probability }
}

/// 64×356 특징 → 단어 후보(확률 내림차순).
public protocol SignClassifying: AnyObject {
    func classify(_ features: [[Float]]) throws -> [GlossScore]
}

public enum TranslationEvent: Equatable, Sendable {
    case wordStarted
    case wordAccepted(GlossScore)
    case wordRejected(GlossScore)
    case classificationFailed(String)
    case sentence(words: [String], text: String)
}

/// realtime.py auto 모드 전체 흐름: 구간 → 분류 → 확신도 필터(앱 추가) → SentenceBuilder.
/// 한 스레드(카메라 큐)에서만 쓴다.
public final class TranslationSession {
    public let stream: SignStream
    public let builder: SentenceBuilder
    public let classifier: SignClassifying
    public var threshold: Double
    public private(set) var lastTopK: [GlossScore] = []

    public var pendingWords: [String] { builder.words }
    public var calibrated: Bool { stream.normalizer.calibrated }

    public init(classifier: SignClassifying, composer: SentenceComposer,
                threshold: Double = RecognitionDefaults.confidenceThreshold,
                style: SentenceStyle = .polite, stream: SignStream = SignStream()) {
        self.classifier = classifier
        self.builder = SentenceBuilder(composer: composer, style: style)
        self.threshold = threshold
        self.stream = stream
    }

    public func process(_ frame: Frame, dt: Double) -> (step: StreamStep, events: [TranslationEvent]) {
        let step = stream.push(frame, dt: dt)
        var events: [TranslationEvent] = []
        if step.label == .started { events.append(.wordStarted) }
        if let seg = step.segment {
            do {
                let top = try classifier.classify(Featurizer.featurize(seg))
                lastTopK = top
                if let best = top.first {
                    if best.probability >= threshold {
                        builder.onWord(best.gloss)
                        events.append(.wordAccepted(best))
                    } else {
                        events.append(.wordRejected(best))
                    }
                }
            } catch {
                events.append(.classificationFailed(String(describing: error)))
            }
        }
        if step.isIdle && builder.tickIdle() {
            let (words, text) = builder.finalize()
            events.append(.sentence(words: words, text: text))
        }
        return (step, events)
    }

    /// 모은 단어로 문장을 바로 완성한다 (realtime.py 의 ENTER).
    public func finalizeNow() -> TranslationEvent? {
        guard !builder.words.isEmpty else { return nil }
        let (words, text) = builder.finalize()
        return .sentence(words: words, text: text)
    }

    public func clearWords() { builder.clear() }
}
```

```swift
public enum TrackingGuidance: Equatable, Sendable {
    case ok
    case calibrating      // 어깨너비 보정 전
    case showUpperBody    // 2초 이상 어깨 미검출
    case showHands        // 3초 이상 양손 미검출
}

/// 촬영 안내 판정 (PRD 화면 5 오류 상태). 원시 frame89(스무딩 전)를 넣는다.
public struct TrackingMonitor: Sendable {
    public var bodyMissingAfter = 2.0
    public var handsMissingAfter = 3.0
    private var bodyMissing = 0.0
    private var handsMissing = 0.0

    public init() {}

    public mutating func update(frame: Frame, calibrated: Bool, dt: Double) -> TrackingGuidance {
        let hasBody = Featurizer.shoulderScale(frame) >= Featurizer.degenerateScalePx
        let hasHands = Layout.leftHand.contains { frame[$0] != .zero }
            || Layout.rightHand.contains { frame[$0] != .zero }
        bodyMissing = hasBody ? 0 : bodyMissing + dt
        handsMissing = hasHands ? 0 : handsMissing + dt
        if !calibrated { return .calibrating }
        if bodyMissing >= bodyMissingAfter - 1e-9 { return .showUpperBody }
        if handsMissing >= handsMissingAfter - 1e-9 { return .showHands }
        return .ok
    }
}
```

```swift
        case bye

    public static let currentVersion = 1

    public var kind: Kind
    public var seq: Int
```

```swift
import Foundation
import OnnxRuntimeBindings
import SignCore

public enum ClassifierError: Error, Equatable {
    case badInputShape(rows: Int, cols: Int)
    case missingOutput
    case missingResource(String)
}

/// sign_c1200.onnx (입력 features (1,64,356), 출력 logits (1,1200)) 래퍼.
/// ORTSession 은 한 스레드(카메라 큐)에서만 쓴다.
public final class SignClassifier: SignClassifying {
    public static let frames = 64
    public let labels: [String]
    private let env: ORTEnv
    private let session: ORTSession

    public init(modelURL: URL, labelsURL: URL) throws {
        env = try ORTEnv(loggingLevel: .warning)
        let options = try ORTSessionOptions()
        try options.setIntraOpNumThreads(1)
        session = try ORTSession(env: env, modelPath: modelURL.path, sessionOptions: options)
        labels = try JSONDecoder().decode([String].self, from: Data(contentsOf: labelsURL))
    }

    public static func bundled() throws -> SignClassifier {
        guard let m = Bundle.module.url(forResource: "sign_c1200", withExtension: "onnx", subdirectory: "Resources")
        else { throw ClassifierError.missingResource("sign_c1200.onnx") }
        guard let l = Bundle.module.url(forResource: "labels", withExtension: "json", subdirectory: "Resources")
        else { throw ClassifierError.missingResource("labels.json") }
        return try SignClassifier(modelURL: m, labelsURL: l)
    }

    public func logits(_ features: [[Float]]) throws -> [Float] {
        guard features.count == Self.frames, features.allSatisfy({ $0.count == Featurizer.featureDim }) else {
            throw ClassifierError.badInputShape(rows: features.count, cols: features.first?.count ?? 0)
        }
        let flat = features.flatMap { $0 }
        let data = flat.withUnsafeBufferPointer {
            NSMutableData(bytes: $0.baseAddress!, length: $0.count * MemoryLayout<Float>.stride)
        }
        let input = try ORTValue(tensorData: data, elementType: .float,
                                 shape: [1, NSNumber(value: Self.frames), NSNumber(value: Featurizer.featureDim)])
        let out = try session.run(withInputs: ["features": input], outputNames: ["logits"], runOptions: nil)
        guard let value = out["logits"] else { throw ClassifierError.missingOutput }
        let bytes = try value.tensorData() as Data
        return bytes.withUnsafeBytes { Array($0.bindMemory(to: Float.self)) }
    }

    public func classify(_ features: [[Float]]) throws -> [GlossScore] {
        let l = try logits(features).map(Double.init)
        let m = l.max() ?? 0
        let e = l.map { Foundation.exp($0 - m) }
        let sum = e.reduce(0, +)
        return Array(Self.mergeByGloss(probabilities: e.map { $0 / sum }, labels: labels).prefix(5))
    }

    /// 같은 한글 단어가 여러 클래스 ID에 있으면(1200개 중 고유 1166개) 확률을 합친다.
    static func mergeByGloss(probabilities: [Double], labels: [String]) -> [GlossScore] {
        var byGloss: [String: Double] = [:]
        for (i, p) in probabilities.enumerated() { byGloss[labels[i], default: 0] += p }
        return byGloss.map { GlossScore(gloss: $0.key, probability: $0.value) }
            .sorted { $0.probability > $1.probability }
    }
}
```

```swift
import Foundation
import SignCore

struct NearbyPeer: Identifiable, Hashable { let id: String; let name: String; let role: UserRole }
struct CaptionLine: Identifiable, Equatable { let id = UUID(); let text: String; let time: Date }
enum Screen: Equatable { case start, discovery, request(NearbyPeer), conversation }
enum LinkState: Equatable { case connected, reconnecting, lost }

protocol PeerTransport: AnyObject {
    func start(name: String, role: UserRole)
    func stop()
    func invite(peerID: String)
    func cancelInvite()
    func respond(accept: Bool)
    func send(_ m: PeerMessage)
    func disconnect()
}

protocol HearingControlling: AnyObject {
    func start()
    func stop()
}

/// 화면 흐름과 대화 상태. 서비스는 이벤트 메서드로만 상태를 바꾼다.
@MainActor final class AppModel: ObservableObject {
    @Published var screen: Screen = .start
    @Published var role: UserRole?
    @Published var myName: String { didSet { defaults.set(myName, forKey: "myName") } }
    @Published var nearby: [NearbyPeer] = []
    @Published var pendingRequest: NearbyPeer?
    @Published var partner: NearbyPeer?
    @Published var link: LinkState = .connected
    @Published var captions: [CaptionLine] = []
    @Published var liveCaption = ""
    @Published var translation = ""
    @Published var translationWords: [String] = []
    @Published var hint: String?
    @Published var toast: String?
    @Published var peerStatus: PeerStatus?
    @Published var devInfo = ""
    @Published var showDevInfo = false

    var transport: PeerTransport?
    var hearing: HearingControlling?
    private let defaults: UserDefaults
    private var seq = 0

    init(defaults: UserDefaults = .standard) {
        self.defaults = defaults
        myName = defaults.string(forKey: "myName") ?? "사용자\(Int.random(in: 100...999))"
    }

    // MARK: 사용자 동작
    func choose(role: UserRole) {
        self.role = role
        screen = .discovery
        nearby = []
        transport?.start(name: myName, role: role)
    }

    func sendRequest(to peer: NearbyPeer) {
        pendingRequest = peer
        transport?.invite(peerID: peer.id)
    }

    func cancelRequest() {
        transport?.cancelInvite()
        pendingRequest = nil
    }

    func accept() { transport?.respond(accept: true) }

    func decline() {
        transport?.respond(accept: false)
        screen = .discovery
    }

    func rename(_ name: String) {
        let trimmed = name.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !trimmed.isEmpty, let role else { return }
        myName = trimmed
        transport?.stop()
        transport?.start(name: myName, role: role)
    }

    func endConversation() {
        send(.bye)
        hearing?.stop()
        transport?.disconnect()
        resetConversation()
        screen = .discovery
    }

    func searchAgain() {
        hearing?.stop()
        transport?.disconnect()
        resetConversation()
        screen = .discovery
        if let role { transport?.stop(); transport?.start(name: myName, role: role) }
    }

    func backToStart() {
        transport?.stop()
        resetConversation()
        nearby = []
        role = nil
        screen = .start
    }

    // MARK: 서비스 이벤트
    func peerFound(_ peer: NearbyPeer) {
        guard let role, peer.role == role.opposite else { return }
        nearby.removeAll { $0.id == peer.id }
        nearby.append(peer)
    }

    func peerLost(id: String) { nearby.removeAll { $0.id == id } }

    func requestReceived(from peer: NearbyPeer) {
        guard screen == .discovery else { transport?.respond(accept: false); return }
        screen = .request(peer)
    }

    func requestFailed(peerID: String) {
        guard let p = pendingRequest, p.id == peerID else { return }
        pendingRequest = nil
        toast = "\(p.name)님이 요청을 거절했어요"
    }

    func requestCancelledByPeer() {
        if case .request = screen {
            screen = .discovery
            toast = "요청이 취소됐어요"
        }
    }

    func connected(to peer: NearbyPeer) {
        pendingRequest = nil
        partner = peer
        link = .connected
        screen = .conversation
        if role == .hearing { hearing?.start() }
    }

    func disconnected(from peerID: String) {
        guard partner?.id == peerID, screen == .conversation else { return }
        hearing?.stop()
        link = .lost
    }

    func received(_ m: PeerMessage) {
        switch m.kind {
        case .captionPartial: liveCaption = m.text ?? ""
        case .captionFinal:
            if let t = m.text, !t.isEmpty { captions.append(CaptionLine(text: t, time: m.sentAt)) }
            liveCaption = ""
        case .translationFinal: translation = m.text ?? ""
        case .status: peerStatus = m.status
        case .hello: break
        case .bye:
            let name = partner?.name ?? "상대방"
            hearing?.stop()
            transport?.disconnect()
            resetConversation()
            screen = .discovery
            toast = "\(name)님이 대화를 끝냈어요"
        }
    }

    // MARK: 청인 쪽 출력 (HearingPipeline/SpeechService 가 호출)
    func send(_ kind: PeerMessage.Kind, text: String? = nil, status: PeerStatus? = nil) {
        seq += 1
        transport?.send(PeerMessage(kind: kind, seq: seq, text: text, status: status))
    }

    private func resetConversation() {
        partner = nil
        pendingRequest = nil
        captions = []
        liveCaption = ""
        translation = ""
        translationWords = []
        hint = nil
        peerStatus = nil
        link = .connected
    }
}
```

```swift
import Foundation
import MultipeerConnectivity
import SignCore

/// MultipeerConnectivity 로 주변 기기 찾기 → 초대 → 수락 → 텍스트 메시지 주고받기.
/// 블루투스와 기기 간 와이파이를 시스템이 알아서 고른다.
final class PeerService: NSObject, PeerTransport {
    static let serviceType = "sign-trans"
    private weak var model: AppModel?
    private var me: MCPeerID?
    private var intro: PeerIntro?
    private var session: MCSession?
    private var advertiser: MCNearbyServiceAdvertiser?
    private var browser: MCNearbyServiceBrowser?
    private var peers: [String: (MCPeerID, NearbyPeer)] = [:]
    private var pendingInvitation: ((Bool, MCSession?) -> Void)?
    private var invitingPeer: MCPeerID?
    private var connectedPeer: NearbyPeer?

    init(model: AppModel) { self.model = model }

    func start(name: String, role: UserRole) {
        stop()
        let id = MCPeerID(displayName: name)
        me = id
        intro = PeerIntro(name: name, role: role)
        let s = MCSession(peer: id, securityIdentity: nil, encryptionPreference: .required)
        s.delegate = self
        session = s
        let adv = MCNearbyServiceAdvertiser(peer: id, discoveryInfo: ["name": name, "role": role.rawValue],
                                            serviceType: Self.serviceType)
        adv.delegate = self
        adv.startAdvertisingPeer()
        advertiser = adv
        let br = MCNearbyServiceBrowser(peer: id, serviceType: Self.serviceType)
        br.delegate = self
        br.startBrowsingForPeers()
        browser = br
    }

    func stop() {
        advertiser?.stopAdvertisingPeer()
        browser?.stopBrowsingForPeers()
        session?.disconnect()
        advertiser = nil; browser = nil; session = nil
        peers = [:]; invitingPeer = nil; pendingInvitation = nil; connectedPeer = nil
    }

    func invite(peerID: String) {
        guard let (mc, _) = peers[peerID], let session, let intro,
              let ctx = try? JSONEncoder().encode(intro) else { return }
        invitingPeer = mc
        browser?.invitePeer(mc, to: session, withContext: ctx, timeout: 30)
    }

    /// MC 에는 초대 취소가 없어 세션을 새로 만든다.
    func cancelInvite() {
        invitingPeer = nil
        resetSession()
    }

    func respond(accept: Bool) {
        pendingInvitation?(accept, accept ? session : nil)
        pendingInvitation = nil
    }

    func send(_ m: PeerMessage) {
        guard let session, !session.connectedPeers.isEmpty, let data = try? m.encoded() else { return }
        try? session.send(data, toPeers: session.connectedPeers, with: .reliable)
    }

    func disconnect() {
        connectedPeer = nil
        resetSession()
    }

    private func resetSession() {
        session?.disconnect()
        guard let me else { return }
        let s = MCSession(peer: me, securityIdentity: nil, encryptionPreference: .required)
        s.delegate = self
        session = s
    }

    private func key(_ p: MCPeerID) -> String { "\(p.displayName)#\(p.hash)" }

    private func onMain(_ f: @escaping @MainActor (AppModel) -> Void) {
        Task { @MainActor [weak model] in if let model { f(model) } }
    }
}

extension PeerService: MCNearbyServiceBrowserDelegate {
    func browser(_ browser: MCNearbyServiceBrowser, foundPeer peerID: MCPeerID, withDiscoveryInfo info: [String: String]?) {
        guard let info, let name = info["name"], let role = info["role"].flatMap(UserRole.init(rawValue:)) else { return }
        let peer = NearbyPeer(id: key(peerID), name: name, role: role)
        peers[peer.id] = (peerID, peer)
        onMain { $0.peerFound(peer) }
    }

    func browser(_ browser: MCNearbyServiceBrowser, lostPeer peerID: MCPeerID) {
        let id = key(peerID)
        peers[id] = nil
        onMain { $0.peerLost(id: id) }
    }
}

extension PeerService: MCNearbyServiceAdvertiserDelegate {
    func advertiser(_ advertiser: MCNearbyServiceAdvertiser, didReceiveInvitationFromPeer peerID: MCPeerID,
                    withContext context: Data?, invitationHandler: @escaping (Bool, MCSession?) -> Void) {
        guard let context, let intro = try? JSONDecoder().decode(PeerIntro.self, from: context) else {
            invitationHandler(false, nil); return
        }
        let peer = NearbyPeer(id: key(peerID), name: intro.name, role: intro.role)
        peers[peer.id] = (peerID, peer)
        pendingInvitation = invitationHandler
        onMain { $0.requestReceived(from: peer) }
    }
}

extension PeerService: MCSessionDelegate {
    func session(_ session: MCSession, peer peerID: MCPeerID, didChange state: MCSessionState) {
        guard session === self.session else { return }
        let id = key(peerID)
        let peer = peers[id]?.1 ?? NearbyPeer(id: id, name: peerID.displayName, role: .deaf)
        switch state {
        case .connected:
            invitingPeer = nil
            connectedPeer = peer
            onMain { $0.connected(to: peer) }
        case .notConnected:
            if invitingPeer == peerID {
                invitingPeer = nil
                onMain { $0.requestFailed(peerID: id) }
            } else if connectedPeer?.id == id {
                connectedPeer = nil
                onMain { $0.disconnected(from: id) }
            } else {
                onMain { $0.requestCancelledByPeer() }
            }
        case .connecting: break
        @unknown default: break
        }
    }

    func session(_ session: MCSession, didReceive data: Data, fromPeer peerID: MCPeerID) {
        guard let m = try? PeerMessage.decode(data) else { return }
        onMain { $0.received(m) }
    }

    func session(_ session: MCSession, didReceive stream: InputStream, withName streamName: String, fromPeer peerID: MCPeerID) {}
    func session(_ session: MCSession, didStartReceivingResourceWithName resourceName: String, fromPeer peerID: MCPeerID, with progress: Progress) {}
    func session(_ session: MCSession, didFinishReceivingResourceWithName resourceName: String, fromPeer peerID: MCPeerID, at localURL: URL?, withError error: Error?) {}
}
```

```swift
import AVFoundation
import Foundation
import Speech
import SignCore

/// 한국어 온디바이스 음성 인식 (iOS 26 SpeechAnalyzer). 부분 결과(volatile)와 최종 결과를 콜백으로 낸다.
final class SpeechService {
    var onPartial: (String) -> Void = { _ in }
    var onFinal: (String) -> Void = { _ in }
    var onStopped: (PeerStatus) -> Void = { _ in }
    var onLog: (String) -> Void = { _ in }

    private let engine = AVAudioEngine()
    private var analyzer: SpeechAnalyzer?
    private var input: AsyncStream<AnalyzerInput>.Continuation?
    private var resultsTask: Task<Void, Never>?

    func start() async {
        guard await AVAudioApplication.requestRecordPermission() else { onStopped(.micOff); return }
        guard let locale = await SpeechTranscriber.supportedLocale(equivalentTo: Locale(identifier: "ko-KR")) else {
            onLog("SpeechTranscriber: 한국어 미지원"); onStopped(.speechStopped); return
        }
        do {
            let transcriber = SpeechTranscriber(locale: locale, transcriptionOptions: [],
                                                reportingOptions: [.volatileResults], attributeOptions: [])
            if let req = try await AssetInventory.assetInstallationRequest(supporting: [transcriber]) {
                onLog("음성 모델 내려받는 중")
                try await req.downloadAndInstall()
            }
            let analyzer = SpeechAnalyzer(modules: [transcriber])
            self.analyzer = analyzer
            guard let format = await SpeechAnalyzer.bestAvailableAudioFormat(compatibleWith: [transcriber]) else {
                onStopped(.speechStopped); return
            }
            let (stream, cont) = AsyncStream.makeStream(of: AnalyzerInput.self)
            input = cont
            resultsTask = Task { [weak self] in
                do {
                    for try await r in transcriber.results {
                        let text = String(r.text.characters)
                        if r.isFinal { self?.onFinal(text) } else { self?.onPartial(text) }
                    }
                } catch {
                    self?.onLog("results error \(error)"); self?.onStopped(.speechStopped)
                }
            }
            try await analyzer.start(inputSequence: stream)
            try startAudio(to: format, cont: cont)
            onLog("음성 인식 시작 \(locale.identifier)")
        } catch {
            onLog("speech start error \(error)")
            onStopped(.speechStopped)
        }
    }

    private func startAudio(to format: AVAudioFormat, cont: AsyncStream<AnalyzerInput>.Continuation) throws {
        let audio = AVAudioSession.sharedInstance()
        try audio.setCategory(.playAndRecord, mode: .measurement, options: [.mixWithOthers, .defaultToSpeaker])
        try audio.setActive(true)
        let node = engine.inputNode
        let inFormat = node.outputFormat(forBus: 0)
        guard let converter = AVAudioConverter(from: inFormat, to: format) else { throw CocoaError(.featureUnsupported) }
        node.installTap(onBus: 0, bufferSize: 4096, format: inFormat) { buffer, _ in
            let ratio = format.sampleRate / inFormat.sampleRate
            let cap = AVAudioFrameCount(Double(buffer.frameLength) * ratio + 1024)
            guard let out = AVAudioPCMBuffer(pcmFormat: format, frameCapacity: cap) else { return }
            var fed = false
            var err: NSError?
            converter.convert(to: out, error: &err) { _, status in
                if fed { status.pointee = .noDataNow; return nil }
                fed = true; status.pointee = .haveData; return buffer
            }
            if err == nil, out.frameLength > 0 { cont.yield(AnalyzerInput(buffer: out)) }
        }
        engine.prepare()
        try engine.start()
    }

    func stop() async {
        engine.inputNode.removeTap(onBus: 0)
        engine.stop()
        input?.finish()
        input = nil
        try? await analyzer?.finalizeAndFinishThroughEndOfInput()
        resultsTask?.cancel()
        analyzer = nil
    }
}
```

```swift
import AVFoundation

enum CameraStartResult { case running, denied, unavailable }

/// 카메라 한 대. 프레임은 queue(직렬)에서 onFrame 으로 나간다.
final class CameraService: NSObject, AVCaptureVideoDataOutputSampleBufferDelegate {
    let session = AVCaptureSession()
    let queue = DispatchQueue(label: "camera.frames")
    var onFrame: (CMSampleBuffer) -> Void = { _ in }
    private var configured = false

    func start(position: AVCaptureDevice.Position, wantFrames: Bool) async -> CameraStartResult {
        guard await AVCaptureDevice.requestAccess(for: .video) else { return .denied }
        guard let device = AVCaptureDevice.default(.builtInWideAngleCamera, for: .video, position: position),
              let input = try? AVCaptureDeviceInput(device: device) else { return .unavailable }
        if !configured {
            session.beginConfiguration()
            session.sessionPreset = .hd1280x720
            if session.canAddInput(input) { session.addInput(input) }
            if wantFrames {
                let out = AVCaptureVideoDataOutput()
                out.videoSettings = [kCVPixelBufferPixelFormatTypeKey as String: kCVPixelFormatType_32BGRA]
                out.alwaysDiscardsLateVideoFrames = true
                out.setSampleBufferDelegate(self, queue: queue)
                if session.canAddOutput(out) { session.addOutput(out) }
                // 세로 화면 그대로의 버퍼를 받는다 (MediaPipe 에 orientation .up 으로 넘김)
                if let c = out.connection(with: .video), c.isVideoRotationAngleSupported(90) { c.videoRotationAngle = 90 }
            }
            session.commitConfiguration()
            configured = true
        }
        queue.async { [session] in session.startRunning() }
        return .running
    }

    func stop() { queue.async { [session] in session.stopRunning() } }

    func captureOutput(_ output: AVCaptureOutput, didOutput sampleBuffer: CMSampleBuffer, from connection: AVCaptureConnection) {
        onFrame(sampleBuffer)
    }
}
```

```swift
/// MediaPipe Holistic (VIDEO 모드, Python realtime.py 와 같은 옵션) → 89점 프레임.
final class HolisticTracker {
    private let landmarker: HolisticLandmarker

    init() throws {
        guard let path = Bundle.main.path(forResource: "holistic_landmarker", ofType: "task") else {
            throw CocoaError(.fileNoSuchFile)
        }
        let o = HolisticLandmarkerOptions()
        o.baseOptions.modelAssetPath = path
        o.runningMode = .video
        o.minPoseDetectionConfidence = 0.5
        o.minHandLandmarksConfidence = 0.3
        landmarker = try HolisticLandmarker(options: o)
    }

    func frame(from buffer: CMSampleBuffer, timestampMs: Int, useFace: Bool) throws -> Frame {
        let image = try MPImage(sampleBuffer: buffer, orientation: .up)
        let r = try landmarker.detect(videoFrame: image, timestampInMilliseconds: timestampMs)
        let pb = CMSampleBufferGetImageBuffer(buffer)!
        let w = Double(CVPixelBufferGetWidth(pb)), h = Double(CVPixelBufferGetHeight(pb))
        func pts(_ l: [NormalizedLandmark]) -> [LandmarkPoint]? {
            l.isEmpty ? nil : l.map { LandmarkPoint(x: Double($0.x), y: Double($0.y)) }
        }
        return MediaPipeAdapter.frame89(pose: pts(r.poseLandmarks), leftHand: pts(r.leftHandLandmarks),
                                        rightHand: pts(r.rightHandLandmarks), face: pts(r.faceLandmarks),
                                        width: w, height: h, useFace: useFace)
    }
}
```

```swift
import AVFoundation
import Foundation
import SignCore
import SignInference

/// 청인 폰: 카메라 → MediaPipe → TranslationSession → 화면/상대 폰.
/// tracker·session 은 camera.queue 에서만 쓴다.
final class HearingPipeline {
    let camera = CameraService()
    private weak var model: AppModel?
    private var tracker: HolisticTracker?
    private var session: TranslationSession?
    private var monitor = TrackingMonitor()
    private var lastTs: Double?
    private var frames = 0, fpsWindowStart = CACurrentMediaTime(), fps = 0.0
    private var lastInferMs = 0.0
    var useFace = true

    init(model: AppModel) { self.model = model }

    func start() {
        Task {
            let result = await camera.start(position: .back, wantFrames: true)
            if result != .running {
                await MainActor.run {
                    model?.hint = result == .denied ? "수어를 번역하려면 카메라 권한이 필요해요" : "카메라를 쓸 수 없어요"
                    model?.send(.status, status: .cameraOff)
                }
                return
            }
            camera.queue.async { [self] in
                do {
                    if tracker == nil { tracker = try HolisticTracker() }
                    if session == nil {
                        session = TranslationSession(classifier: try SignClassifier.bundled(), composer: try .bundled())
                    }
                } catch {
                    DispatchQueue.main.async { self.model?.hint = "인식 모델을 불러오지 못했어요" }
                }
            }
            camera.onFrame = { [weak self] buf in self?.handle(buf) }
        }
    }

    func stop() {
        camera.stop()
        camera.queue.async { [self] in session?.clearWords(); lastTs = nil }
    }

    /// camera.queue 에서 호출된다.
    private func handle(_ buf: CMSampleBuffer) {
        guard let tracker, let session else { return }
        let t = CMSampleBufferGetPresentationTimeStamp(buf).seconds
        let dt = lastTs.map { t - $0 } ?? (1.0 / 30)
        lastTs = t
        guard let frame = try? tracker.frame(from: buf, timestampMs: Int(t * 1000), useFace: useFace) else { return }
        let t0 = CACurrentMediaTime()
        let (step, events) = session.process(frame, dt: dt)
        if step.segment != nil { lastInferMs = (CACurrentMediaTime() - t0) * 1000 }
        let guidance = monitor.update(frame: frame, calibrated: session.calibrated, dt: dt)
        frames += 1
        if t0 - fpsWindowStart >= 1 { fps = Double(frames) / (t0 - fpsWindowStart); frames = 0; fpsWindowStart = t0 }
        let words = session.pendingWords
        let top = session.lastTopK.prefix(3).map { "\($0.gloss) \(Int($0.probability * 100))%" }.joined(separator: ", ")
        let dev = String(format: "fps %.1f  e %.3f  %@\n추론(구간포함) %.1fms\n%@", fps, step.energy,
                         step.isIdle ? "대기" : "녹화중", lastInferMs, top)
        DispatchQueue.main.async { [weak self] in
            guard let m = self?.model else { return }
            m.devInfo = dev
            m.translationWords = words
            m.hint = switch guidance {
            case .calibrating: "상반신과 양 어깨가 보이게 비춰 주세요"
            case .showUpperBody: "상반신과 양 어깨가 화면에 들어오게 해 주세요"
            case .showHands: "손이 화면 안에 보이게 해 주세요"
            case .ok: nil
            }
            for e in events {
                switch e {
                case .wordRejected: m.hint = "다시 해 주세요"
                case .sentence(_, let text):
                    m.translation = text
                    m.send(.translationFinal, text: text)
                default: break
                }
            }
        }
    }
}
```
